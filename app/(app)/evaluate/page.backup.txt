
'use client';

import { useEffect, useState } from 'react';
import { useApp } from '@/components/Providers';
import { Banner, Fld } from '@/components/ui';
import { createClient } from '@/lib/supabase';

export default function Evaluate() {
  const { toast } = useApp();

  const [submissions, setSubmissions] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [savingId, setSavingId] = useState(null);

  const [forms, setForms] = useState({});

  useEffect(() => {
    let active = true;

    async function loadSubmissions() {
      const supabase = createClient();

      // Load real accomplishment submissions.
      const { data, error: submissionError } = await supabase
        .from('work_submissions')
        .select('id, task_id, employee_id, accomplishment, submitted_at')
        .order('submitted_at', { ascending: false });

      if (!active) return;

      if (submissionError) {
        console.error('Load submissions error:', submissionError);
        setError(submissionError.message);
        setLoading(false);
        return;
      }

      const records = data || [];

      const taskIds = [...new Set(records.map((s) => s.task_id))];
      const employeeIds = [
        ...new Set(records.map((s) => s.employee_id)),
      ];

      // Load task names.
      let taskNames = {};

      if (taskIds.length > 0) {
        const { data: tasks, error: taskError } = await supabase
          .from('tasks')
          .select('id, title')
          .in('id', taskIds);

        if (taskError) {
          console.error('Load task names error:', taskError);
          setError(taskError.message);
          setLoading(false);
          return;
        }

        taskNames = Object.fromEntries(
          (tasks || []).map((task) => [task.id, task.title])
        );
      }

      // Load employee names.
      let employeeNames = {};

      if (employeeIds.length > 0) {
        const { data: employees, error: employeeError } =
          await supabase
            .from('profiles')
            .select('id, full_name')
            .in('id', employeeIds);

        if (employeeError) {
          console.error('Load employee names error:', employeeError);
          setError(employeeError.message);
          setLoading(false);
          return;
        }

        employeeNames = Object.fromEntries(
          (employees || []).map((employee) => [
            employee.id,
            employee.full_name,
          ])
        );
      }

      // Check which submissions already have evaluations.
      let evaluatedIds = new Set();

      if (records.length > 0) {
        const { data: evaluations, error: evaluationError } =
          await supabase
            .from('evaluations')
            .select('submission_id')
            .in(
              'submission_id',
              records.map((record) => record.id)
            );

        if (evaluationError) {
          console.error('Load evaluations error:', evaluationError);
          setError(evaluationError.message);
          setLoading(false);
          return;
        }

        evaluatedIds = new Set(
          (evaluations || []).map((item) => item.submission_id)
        );
      }

      if (!active) return;

      setSubmissions(
        records.map((record) => ({
          ...record,
          taskName: taskNames[record.task_id] || 'Unknown task',
          employeeName:
            employeeNames[record.employee_id] || 'Employee',
          evaluated: evaluatedIds.has(record.id),
        }))
      );

      setError('');
      setLoading(false);
    }

    loadSubmissions();

    return () => {
      active = false;
    };
  }, []);

  function updateForm(submissionId, field, value) {
    setForms((previous) => ({
      ...previous,
      [submissionId]: {
        rating: 5,
        feedback: '',
        ...previous[submissionId],
        [field]: value,
      },
    }));
  }

  async function submitEvaluation(submissionId) {
    if (savingId) return;

    const form = forms[submissionId] || {
      rating: 5,
      feedback: '',
    };

    setSavingId(submissionId);

    try {
      const supabase = createClient();

      const { error: evaluationError } = await supabase.rpc(
        'evaluate_work',
        {
          target_submission_id: submissionId,
          new_rating: Number(form.rating),
          new_feedback: form.feedback,
        }
      );

      if (evaluationError) throw evaluationError;

      setSubmissions((previous) =>
        previous.map((submission) =>
          submission.id === submissionId
            ? { ...submission, evaluated: true }
            : submission
        )
      );

      toast('Evaluation saved successfully');
    } catch (evaluationError) {
      console.error('Evaluate work error:', evaluationError);
      alert(evaluationError.message);
    } finally {
      setSavingId(null);
    }
  }

  function formatDate(value) {
    if (!value) return 'Unknown date';

    return new Date(value).toLocaleDateString('en-PH', {
      year: 'numeric',
      month: 'short',
      day: 'numeric',
    });
  }

  return (
    <>
      <Banner
        title="Evaluate accomplishments"
        sub="Review submitted work and update performance records."
      />

      {loading ? (
        <div className="card">
          <p>Loading employee submissions...</p>
        </div>
      ) : error ? (
        <div className="card">
          <p style={{ color: 'red' }}>{error}</p>
        </div>
      ) : submissions.length === 0 ? (
        <div className="card">
          <h3>No submissions yet</h3>
          <p className="muted">
            Employee accomplishments will appear here after submission.
          </p>
        </div>
      ) : (
        submissions.map((submission) => {
          const form = forms[submission.id] || {
            rating: 5,
            feedback: '',
          };

          return (
            <div
              className="card"
              key={submission.id}
              style={{ marginBottom: 18 }}
            >
              <div className="row sb wrap">
                <h3 style={{ margin: 0 }}>
                  {submission.taskName}
                </h3>

                <span
                  className={
                    submission.evaluated
                      ? 'chip ok'
                      : 'chip pd'
                  }
                >
                  {submission.evaluated
                    ? 'Evaluated'
                    : 'Needs review'}
                </span>
              </div>

              <p className="muted sm">
                {submission.employeeName} submitted{' '}
                {formatDate(submission.submitted_at)}
              </p>

              <h4>Accomplishment</h4>

              <p className="muted">
                {submission.accomplishment}
              </p>

              {!submission.evaluated && (
                <>
                  <Fld label="Rating">
                    <select
                      className="in"
                      value={form.rating}
                      onChange={(e) =>
                        updateForm(
                          submission.id,
                          'rating',
                          Number(e.target.value)
                        )
                      }
                    >
                      <option value={5}>5 - Excellent</option>
                      <option value={4}>4 - Very good</option>
                      <option value={3}>3 - Good</option>
                      <option value={2}>2 - Needs improvement</option>
                      <option value={1}>1 - Poor</option>
                    </select>
                  </Fld>

                  <Fld label="Feedback">
                    <textarea
                      className="in"
                      placeholder="Add remarks for the employee"
                      value={form.feedback}
                      onChange={(e) =>
                        updateForm(
                          submission.id,
                          'feedback',
                          e.target.value
                        )
                      }
                    />
                  </Fld>

                  <div className="row wrap">
                    <button
                      className="btn p"
                      disabled={savingId !== null}
                      onClick={() =>
                        submitEvaluation(submission.id)
                      }
                    >
                      {savingId === submission.id
                        ? 'Saving...'
                        : 'Save evaluation'}
                    </button>
                  </div>
                </>
              )}
            </div>
          );
        })
      )}
    </>
  );
}
